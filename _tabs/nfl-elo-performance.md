<script>
(() => {
  "use strict";

  const dataUrl = "{{ '/assets/data/nfl-elo/performance_games.csv' | relative_url }}";
  const epsilon = 1e-15;

  function parseCsv(text) {
    const records = [];
    let record = [];
    let field = "";
    let quoted = false;

    for (let index = 0; index < text.length; index += 1) {
      const character = text[index];
      if (quoted) {
        if (character === '"' && text[index + 1] === '"') {
          field += '"';
          index += 1;
        } else if (character === '"') {
          quoted = false;
        } else {
          field += character;
        }
      } else if (character === '"') {
        quoted = true;
      } else if (character === ",") {
        record.push(field);
        field = "";
      } else if (character === "\n") {
        record.push(field.replace(/\r$/, ""));
        records.push(record);
        record = [];
        field = "";
      } else {
        field += character;
      }
    }

    if (field || record.length) {
      record.push(field.replace(/\r$/, ""));
      records.push(record);
    }

    const [headers, ...rows] = records;
    return rows
      .filter((row) => row.some((value) => value !== ""))
      .map((row) => Object.fromEntries(headers.map((header, index) => [header, row[index] ?? ""])));
  }

  function sortValues(values) {
    return [...new Set(values)].sort((left, right) => {
      const leftNumber = Number(left);
      const rightNumber = Number(right);
      return Number.isFinite(leftNumber) && Number.isFinite(rightNumber)
        ? leftNumber - rightNumber
        : String(left).localeCompare(String(right), undefined, { numeric: true });
    });
  }

  function addOptions(select, values) {
    values.forEach((value) => {
      const option = document.createElement("option");
      option.value = value;
      option.textContent = value;
      select.append(option);
    });
  }

  function metrics(rows) {
    const totals = rows.reduce((result, row) => {
      const probability = row.pHome;
      const actual = row.actualHomeWin;
      const protectedProbability = Math.min(1 - epsilon, Math.max(epsilon, probability));
      result.correct += Number((probability >= 0.5 ? 1 : 0) === actual);
      result.logLoss += -(
        actual * Math.log(protectedProbability) +
        (1 - actual) * Math.log(1 - protectedProbability)
      );
      result.brier += (probability - actual) ** 2;
      return result;
    }, { correct: 0, logLoss: 0, brier: 0 });

    return {
      games: rows.length,
      accuracy: totals.correct / rows.length,
      logLoss: totals.logLoss / rows.length,
      brier: totals.brier / rows.length,
    };
  }

  function renderRow(table, emptyMessage, labels, rows) {
    const body = table.querySelector("tbody");
    body.replaceChildren();
    const hasRows = rows.length > 0;
    table.closest(".table-wrapper").classList.toggle("d-none", !hasRows);
    emptyMessage.classList.toggle("d-none", hasRows);
    if (!hasRows) return;

    const result = metrics(rows);
    const tr = document.createElement("tr");
    [...labels, result.games, `${(result.accuracy * 100).toFixed(1)}%`, result.logLoss.toFixed(3), result.brier.toFixed(3)]
      .forEach((value, index) => {
        const cell = document.createElement("td");
        cell.textContent = value;
        if (index >= labels.length) cell.classList.add("text-end");
        tr.append(cell);
      });
    body.append(tr);
  }

  fetch(dataUrl)
    .then((response) => {
      if (!response.ok) throw new Error(`HTTP ${response.status}`);
      return response.text();
    })
    .then((text) => {
      const rows = parseCsv(text).map((row) => ({
        season: row.season,
        week: row.week,
        pHome: Number(row.p_home),
        actualHomeWin: Number(row.actual_home_win),
        evaluationType: row.evaluation_type,
      })).filter((row) =>
        row.season && row.week && Number.isFinite(row.pHome) &&
        Number.isFinite(row.actualHomeWin) &&
        (row.evaluationType === "historical" || row.evaluationType === "deployed")
      );

      const historicalRows = rows.filter((row) => row.evaluationType === "historical");
      const deployedRows = rows.filter((row) => row.evaluationType === "deployed");
      const historicalSeason = document.querySelector("#historical-season");
      const historicalWeek = document.querySelector("#historical-week");
      const deployedWeek = document.querySelector("#deployed-week");

      addOptions(historicalSeason, sortValues(historicalRows.map((row) => row.season)));
      addOptions(historicalWeek, sortValues(historicalRows.map((row) => row.week)));
      addOptions(deployedWeek, sortValues(deployedRows.map((row) => row.week)));

      function renderHistorical() {
        const season = historicalSeason.value;
        const week = historicalWeek.value;
        const filtered = historicalRows.filter((row) =>
          (!season || row.season === season) && (!week || row.week === week)
        );
        renderRow(
          document.querySelector("#historical-performance"),
          document.querySelector("#historical-empty"),
          [season || "All Seasons", week || "All Weeks"],
          filtered
        );
      }

      function renderDeployed() {
        const week = deployedWeek.value;
        const filtered = deployedRows.filter((row) => !week || row.week === week);
        renderRow(
          document.querySelector("#deployed-performance"),
          document.querySelector("#deployed-empty"),
          [week || "All Weeks"],
          filtered
        );
      }

      historicalSeason.addEventListener("change", renderHistorical);
      historicalWeek.addEventListener("change", renderHistorical);
      deployedWeek.addEventListener("change", renderDeployed);
      renderHistorical();
      renderDeployed();
    })
    .catch(() => {
      document.querySelector("#historical-performance").closest(".table-wrapper").classList.add("d-none");
      document.querySelector("#deployed-performance").closest(".table-wrapper").classList.add("d-none");
      document.querySelector("#performance-load-error").classList.remove("d-none");
    });
})();
</script>
